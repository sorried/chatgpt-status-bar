cmake_minimum_required(VERSION 3.21)
project(GptStatusBar VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
if(MSVC)
    add_compile_options(/utf-8 /W4)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
add_library(status_core src/session.cpp src/watcher.cpp src/codex.cpp)
target_include_directories(status_core PUBLIC src)
target_link_libraries(status_core PUBLIC Qt6::Widgets)
add_library(status_ui src/app.cpp src/settings.cpp src/activity.cpp src/mascot.cpp)
target_link_libraries(status_ui PUBLIC status_core)
qt_add_executable(gpt-status-bar WIN32 MACOSX_BUNDLE src/main.cpp assets/resources.qrc)
target_link_libraries(gpt-status-bar PRIVATE status_ui)
if(WIN32)
    enable_language(RC)
    configure_file(packaging/icon.rc.in ${CMAKE_CURRENT_BINARY_DIR}/icon.rc @ONLY)
    target_sources(gpt-status-bar PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/icon.rc)
elseif(APPLE)
    set_source_files_properties(assets/icon.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    target_sources(gpt-status-bar PRIVATE assets/icon.icns)
    set_target_properties(gpt-status-bar PROPERTIES MACOSX_BUNDLE_ICON_FILE icon.icns)
endif()
option(DEPLOY_QT "Bundle Qt runtime dependencies on install" OFF)
if(DEPLOY_QT)
    qt_generate_deploy_app_script(TARGET gpt-status-bar OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
endif()
set_target_properties(gpt-status-bar PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "GPT status bar" MACOSX_BUNDLE_GUI_IDENTIFIER "io.github.sorried.gpt-status-bar" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
include(GNUInstallDirs)
configure_file(packaging/version.in ${CMAKE_CURRENT_BINARY_DIR}/version @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/version DESTINATION ${CMAKE_INSTALL_DATADIR}/gpt-status-bar)
install(TARGETS gpt-status-bar BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(DEPLOY_QT)
    install(SCRIPT ${deploy_script})
endif()
install(FILES LICENSE docs/third-party.md docs/licenses/LGPL-3.0-only.txt docs/licenses/GPL-3.0-only.txt DESTINATION licenses)
if(UNIX AND NOT APPLE)
    set_target_properties(gpt-status-bar PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
endif()
if(UNIX AND NOT APPLE)
    install(FILES packaging/io.github.sorried.gpt-status-bar.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
    install(FILES assets/chatgpt.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/1024x1024/apps RENAME io.github.sorried.gpt-status-bar.png)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(status-tests tests/test_core.cpp)
    target_link_libraries(status-tests PRIVATE status_core Qt6::Test)
    add_test(NAME status-core COMMAND status-tests)
    set_tests_properties(status-core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    qt_add_executable(preferences-tests tests/test_preferences.cpp assets/resources.qrc)
    target_link_libraries(preferences-tests PRIVATE status_ui Qt6::Test)
    add_test(NAME preferences COMMAND preferences-tests)
    set_tests_properties(preferences PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
        add_test(NAME linux-installer COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_install.sh ${CMAKE_CURRENT_SOURCE_DIR}/install-linux.sh ${CMAKE_CURRENT_BINARY_DIR}/install-test)
    endif()
endif()
set(CPACK_PACKAGE_NAME "gpt-status-bar")
set(CPACK_PACKAGE_VENDOR "sorried")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
include(CPack)
